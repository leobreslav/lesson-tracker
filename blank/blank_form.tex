% Универсальный бланк для распознавания.
%
% Печатается пачкой на весь год и лежит в кабинете: ученик берёт чистый лист.
% Ничего индивидуального на нём нет — ни имени, ни номера работы, ни QR, — а
% значит всё, что система про лист знает, она берёт из шапки: рукописного
% имени и клеток с баллами, которые ставит учитель.
%
% Геометрия (миллиметры от края A4) — единственное, что связывает этот файл с
% кодом чтения: сдвинули метку здесь, кроп на сервере уедет мимо. Поэтому
% числа ниже обязаны совпадать с модулем геометрии в приложении, и правятся
% они вместе.
%
% Компиляция: pdflatex blank_form.tex  (xelatex тоже подойдёт)
%
% Взято из соседнего проекта ScanAnalize и изменено в трёх местах:
%   * реперы вокруг ШАПКИ, а не только по углам листа — см. ниже;
%   * жирная рамка вокруг поля тетрадных клеток;
%   * жирная внешняя рамка у сетки баллов.

\usepackage[a4paper,margin=12mm]{geometry}
% helvet вместо fontspec+Arial: бланк целиком английский, а pdflatex стоит
% всюду и не требует системного шрифта.

% ---- реперы ----------------------------------------------------------------
% Меток шесть, а наборов два, и второй достраивается из первого.
%
% ЧЕТЫРЕ ПО УГЛАМ ЛИСТА (4мм, нижний правый 2.5мм) — «лист целиком попал в
% кадр». Выпрямлять по ним всю страницу мы не будем: PDF приходит и из
% телефонных фотографий, а лист в руке не плоский, и одна гомография на весь
% лист размазывает ошибку по странице — включая ту полосу в 19мм, которую мы
% единственную и читаем.
%
% ДВЕ ПОД ШАПКОЙ, на тех же вертикалях (8 и 202мм) — вместе с верхними
% угловыми они и образуют рамку выпрямления: 194 x 31мм. На такой области лист
% плоский почти всегда, а фотография с обрезанным низом остаётся годной.
% Своих меток над шапкой заводить не нужно — верхние угловые уже стоят там,
% где надо, и второй ряд квадратиков рядом с ними был бы шумом.
%
% Наборы различаются не размером, а пропорцией прямоугольника: у листа это
% портрет (0.69), у шапки — широкая полоса (6.3). Правая метка в каждом наборе
% меньше левой — по ней определяется ориентация, без перебора поворотов.
\newcommand{\fiducials}{%
\begin{tikzpicture}[remember picture,overlay]
 % углы листа
 \fill[black] ([shift={( 6mm,-6mm)}]current page.north west) rectangle ++( 4mm,-4mm);
 \fill[black] ([shift={(-6mm,-6mm)}]current page.north east) rectangle ++(-4mm,-4mm);
 \fill[black] ([shift={( 6mm, 6mm)}]current page.south west) rectangle ++( 4mm, 4mm);
 \fill[black] ([shift={(-6mm, 6mm)}]current page.south east) rectangle ++(-2.5mm, 2.5mm);
 % под шапкой: центры (8, 39) и (202, 39) — те же вертикали, что у верхних
 \fill[black] ([shift={( 6.00mm,-37.00mm)}]current page.north west) rectangle ++( 4mm,-4mm);
 \fill[black] ([shift={(-6.75mm,-37.75mm)}]current page.north east) rectangle ++(-2.5mm,-2.5mm);
\end{tikzpicture}}

% ---- сетка баллов: 15 клеток Q1..Q15 + клетка суммы страницы ----------------
% Ширина = ширине поля тетрадных клеток: 16 равных ячеек по 11.5625мм = 185мм.
% Левый край сдвинут на 0.5мм, чтобы совпасть с началом сетки (12.5мм).
% Внешняя рамка жирнее внутренних линий: по ней чтение уточняет выпрямление
% уже после реперов — линии напечатаны и так, чернил это не стоит, а точность
% даёт в доли миллиметра.
\newcommand{\markgrid}{%
\hspace*{0.5mm}%
\begin{tikzpicture}[x=1mm,y=1mm,line width=0.4pt]
 \def\w{11.5625}\def\hh{16.5}\def\lab{5}
 \foreach \q in {1,...,15}{%
   \pgfmathsetmacro\xx{(\q-1)*\w}%
   \fill[black!7] (\xx,\hh-\lab) rectangle (\xx+\w,\hh);
   \draw (\xx,0) rectangle (\xx+\w,\hh);
   \draw (\xx,\hh-\lab) -- (\xx+\w,\hh-\lab);
   \node[font=\scriptsize\bfseries,text depth=0pt,inner sep=0pt] at (\xx+\w/2,\hh-\lab/2) {Q\q};
 }
 \pgfmathsetmacro\sx{15*\w}
 \fill[black!7] (\sx,\hh-\lab) rectangle (\sx+\w,\hh);
 \draw (\sx,0) rectangle (\sx+\w,\hh);
 \draw (\sx,\hh-\lab) -- (\sx+\w,\hh-\lab);
 \node[font=\scriptsize\bfseries,text depth=0pt,inner sep=0pt] at (\sx+\w/2,\hh-\lab/2) {$\Sigma$ pg};
 \draw[line width=0.8pt] (0,0) rectangle (16*\w,\hh);
\end{tikzpicture}}

% ---- фон: тетрадная сетка 5x5 мм -------------------------------------------
% Сетка ровно из целых клеток: 37 x 47 по 5 мм (185 x 235 мм), привязана к
% началу области (12.5мм / 45мм) — на всех краях полноценные клетки.
% Рамка вокруг неё жирная и почти чёрная: она и отделяет поле записи от полей
% листа, и служит третьим ориентиром, если фотография обрезала угол.
\newcommand{\notebookgrid}{%
\begin{tikzpicture}[remember picture,overlay]
 \coordinate (gtl) at ([shift={(12.5mm,-45mm)}]current page.north west);
 \begin{scope}[shift={(gtl)}]
   \clip (0,0) rectangle (185mm,-230mm);
   \draw[step=5mm,black!40,line width=0.3pt] (0,0) grid (185mm,-230mm);
 \end{scope}
 % рамка рисуется вне clip: иначе её внешняя половина срезалась бы
 \begin{scope}[shift={(gtl)}]
   \draw[black!85,line width=0.9pt] (0,0) rectangle (185mm,-230mm);
 \end{scope}
\end{tikzpicture}}
