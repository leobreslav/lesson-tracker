% ---- реперы ----------------------------------------------------------------
% Шесть меток, все одинаковые (4мм), по три в каждом поле листа: на 8, 27 и
% 39мм от верха плюс угловые внизу страницы. Вертикали две — 8 и 202мм.
%
% Зачем именно так:
%
% ВЫПРЯМЛЯЕТСЯ ШАПКА, А НЕ ЛИСТ. PDF приходит и из телефонных фотографий, а
% лист в руке не плоский: одна гомография на весь лист размазывает ошибку по
% странице — включая ту полосу в 19мм, которую мы единственную и читаем.
% Метки на 8, 27 и 39мм обнимают шапку, и на такой области лист плоский почти
% всегда.
%
% ИХ ШЕСТЬ, А НЕ ЧЕТЫРЕ, потому что рвётся угол. Скрепку ставят в левый
% верхний, и вырывают её вместе с бумагой. Четырёх точек ровно хватает на
% гомографию, то есть потеря любой означает «не собралось»; шесть дают запас,
% и подгонка идёт по всем найденным, а выпадающие отбрасываются.
%
% РАЗМЕР У ВСЕХ ОДИН. Меньшая метка была бы вчетверо беднее пикселями и
% терялась бы первой на смазанном фото — а держалась на ней ориентация.
% Ориентацию теперь говорят расположение (три метки кучно вверху, одна внизу),
% серая полоса Q1..Sigma над клетками и сам QR внизу листа.
%
% Настоящая же опора выпрямления — сетка баллов: семнадцать вертикалей на базе
% 185мм стоят ровно там, где мы кропаем, и переживают и порванный угол, и
% бледную печать. Метки — приближение и проверка «лист целиком в кадре».

% ---- сетка баллов: 15 клеток Q1..Q15 + клетка суммы страницы ----------------
% Ширина = ширине поля тетрадных клеток: 16 равных ячеек по 11.5625мм = 185мм.
% Левый край сдвинут на 0.5мм, чтобы совпасть с началом сетки (12.5мм).
% Внешняя рамка жирнее внутренних линий: по ней чтение уточняет выпрямление
% уже после реперов — линии напечатаны и так, чернил это не стоит, а точность
% даёт в доли миллиметра.
\newcommand{\markgrid}{%
\hspace*{0.5mm}%
\begin{tikzpicture}[x=1mm,y=1mm,line width=0.4pt]
 \def\w{11.5625}\def\hh{16.5}\def\lab{5}
 \foreach \q in {1,...,15}{%
   \pgfmathsetmacro\xx{(\q-1)*\w}%
   \fill[black!7] (\xx,\hh-\lab) rectangle (\xx+\w,\hh);
   \draw (\xx,0) rectangle (\xx+\w,\hh);
   \draw (\xx,\hh-\lab) -- (\xx+\w,\hh-\lab);
   \node[font=\scriptsize\bfseries,text depth=0pt,inner sep=0pt] at (\xx+\w/2,\hh-\lab/2) {Q\q};
 }
 \pgfmathsetmacro\sx{15*\w}
 \fill[black!7] (\sx,\hh-\lab) rectangle (\sx+\w,\hh);
 \draw (\sx,0) rectangle (\sx+\w,\hh);
 \draw (\sx,\hh-\lab) -- (\sx+\w,\hh-\lab);
 \node[font=\scriptsize\bfseries,text depth=0pt,inner sep=0pt] at (\sx+\w/2,\hh-\lab/2) {$\Sigma$ pg};
 \draw[line width=0.8pt] (0,0) rectangle (16*\w,\hh);
\end{tikzpicture}}

% ---- фон: тетрадная сетка 5x5 мм -------------------------------------------
% Сетка ровно из целых клеток: 37 x 47 по 5 мм (185 x 235 мм), привязана к
% началу области (12.5мм / 45мм) — на всех краях полноценные клетки.
% Рамка вокруг неё жирная и почти чёрная: она и отделяет поле записи от полей
% листа, и служит третьим ориентиром, если фотография обрезала угол.
\newcommand{\notebookgrid}{%
\begin{tikzpicture}[remember picture,overlay]
 \coordinate (gtl) at ([shift={(12.5mm,-45mm)}]current page.north west);
 \begin{scope}[shift={(gtl)}]
   \clip (0,0) rectangle (185mm,-230mm);
   \draw[step=5mm,black!40,line width=0.3pt] (0,0) grid (185mm,-230mm);
 \end{scope}
 % белая подложка под QR: без неё тетрадные линии лезли бы в модули, а
 % «тихая зона» в 2мм у кода обязана быть пустой
 \begin{scope}[shift={(gtl)}]
   \fill[white] (167mm,-212mm) rectangle (184mm,-229mm);
 \end{scope}
 % рамка рисуется вне clip и последней: иначе её внешняя половина срезалась бы,
 % а подложка съела бы угол
 \begin{scope}[shift={(gtl)}]
   \draw[black!85,line width=0.9pt] (0,0) rectangle (185mm,-230mm);
 \end{scope}
\end{tikzpicture}%
